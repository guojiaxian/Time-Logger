function checksumOf(arr, keyFn, algo) {
  const str = arr.map(keyFn).join(';');
  if (algo === 'djb2') {
    let h = 5381;
    for (let i = 0; i < str.length; i++) h = ((h << 5) + h + str.charCodeAt(i)) | 0;
    return String(h >>> 0);
  }
  return crypto.createHash('sha256').update(str, 'utf8').digest('hex');
}

function serveStatic(req, res) {
  let p = decodeURIComponent((req.url || '/').split('?')[0]);
  if (p === '/') p = '/index.html';
  const file = path.normalize(path.join(WEB_DIR, p));
  if (!file.startsWith(WEB_DIR + path.sep) && file !== WEB_DIR) {
    res.writeHead(403);

    if (err) {
      res.writeHead(404, { 'Content-Type': 'text/plain; charset=utf-8' });
      return res.end('404 Not Found');
    }
    res.writeHead(200, {
      'Content-Type': MIME[path.extname(file).toLowerCase()] || 'application/octet-stream',
      'Cache-Control': 'no-cache',

  });
}

# 已在运行则退出
if [ -f "$PID_FILE" ] && kill -0 "$(cat "$PID_FILE")" 2>/dev/null; then
  echo "[错误] 服务器已在运行, PID=$(cat "$PID_FILE")"
  echo "如需重启请先执行: ./stop.sh"
  exit 1
fi
# 清理残留 pid 文件
rm -f "$PID_FILE"
echo "启动时间记录器服务器, 端口: $PORT"
echo "日志: $(pwd)/$LOG_FILE"
PORT="$PORT" nohup node server.js > "$LOG_FILE" 2>&1 &
echo $! > "$PID_FILE"
# 等待 2 秒检查是否启动成功
sleep 2
if kill -0 "$(cat "$PID_FILE")" 2>/dev/null; then
  echo "启动成功 (PID=$(cat "$PID_FILE"))"
  echo "HTTPS: https://localhost:$PORT"
  echo "停止:  ./stop.sh"
else
  echo "[失败] 服务器未能启动, 查看日志: tail -30 $LOG_FILE"
  rm -f "$PID_FILE"
  exit 1
fi
